Step B: Run inference on sample data
---------
python3 << 'PYSCRIPT'
import json
import numpy as np
from devicedatahub_ml_engine.models.io import load_model
from devicedatahub_ml_engine.utils.data import ML_FEATURE_COLUMNS

# Load trained model
model_wrapper = load_model("model_artifacts/model.pkl")
model = model_wrapper['model']
scaler = model_wrapper['scaler']
feature_columns = model_wrapper['feature_columns']

# Load sample data
with open('sample_inference_row.json', 'r') as f:
    data = json.load(f)

sample_row = data['rows'][0]
device_id = sample_row.get('device_id', 'unknown')

print(f"\n{'='*70}")
print(f"SAMPLE INFERENCE")
print(f"{'='*70}")
print(f"\n📱 Device: {device_id}")
print(f"📊 Metrics:")
print(f"   - Channel utilization: {sample_row.get('channel_utilization_pct', 0)}%")
print(f"   - TX retries: {sample_row.get('tx_retries', 0)}")
print(f"   - AVG RSSI: {sample_row.get('avg_rssi_dbm', 0)} dBm")
print(f"   - Clients: {sample_row.get('active_client_count', 0)}")
print(f"   - Noise floor: {sample_row.get('noise_floor_dbm', 0)} dBm")

# Extract features in exact order
features = []
for col in feature_columns:
    val = sample_row.get(col, 0.0)
    if val is None:
        val = 0.0
    features.append(float(val))

X = np.array([features])

# Apply scaler
X_scaled = scaler.transform(X)

# Get prediction
anomaly_prob = model.predict_proba(X_scaled)[0, 1]
prediction = model.predict(X_scaled)[0]

print(f"\n🎯 PREDICTION RESULT:")
print(f"   - Anomaly probability: {anomaly_prob:.4f} ({anomaly_prob*100:.2f}%)")
print(f"   - Status: {'🚨 ANOMALY' if anomaly_prob >= 0.5 else '✅ HEALTHY'}")
print(f"   - Model prediction: {prediction} (0=healthy, 1=anomaly)")

print(f"\n📋 Model Info:")
print(f"   - Features used: {len(feature_columns)}")
print(f"   - Alert threshold: 0.5")
print(f"   - Training accuracy: {model_wrapper.get('training_info', {}).get('accuracy', 'N/A')}")
print(f"\n{'='*70}\n")
